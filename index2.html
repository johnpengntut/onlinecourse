診斷 1. 為什麼原版 server4_2.c 需要重構？

原始的 server4_2.c 採用「每連入一個客戶端就 fork() 一個子行程處理」的並行模型，並僅用一行 signal(SIGCHLD, SIG_IGN) 處理子行程結束。在教學示範中雖能運作，但在實際系統環境下存在四個關鍵工程問題：

問題面向 原版 server4_2.c 寫法 潛在風險與後果 改寫後的解決方案
訊號註冊介面 使用傳統 signal(SIGCHLD, SIG_IGN) 早期 Unix 系統行為不一致，且忽略狀態無法記錄子行程退出事件或自訂清理動作。 改用 POSIX 標準 sigaction() 搭配 SA_NOCLDSTOP 與迴圈回收。
伺服器關閉機制 無條件 while(1)，未處理 SIGINT / SIGTERM 按下 Ctrl+C 或收到系統關機訊號時行程直接暴斃，來不及釋放監聽埠與收尾。 註冊 on_stop Handler 設定 volatile sig_atomic_t stop_server 優雅跳出。
競態條件（Race） 若直接檢查旗標後呼叫阻塞 accept() 訊號若在「檢查旗標後、進入 accept() 前」抵達，會發生遺失喚醒（Lost Wakeup）導致卡死。 用 sigprocmask(SIG_BLOCK) 保護旗標檢查，並以 pselect() 原子解除封鎖等待。
檔案描述符洩漏 子行程進入工作前未關閉 server_sockfd 每個子行程都持有監聽 Socket 的參考計數，若父行程重啟或結束，監聽埠不會真正釋放。 子行程在 fork() == 0 後第一時間執行 close(server_sockfd) 並還原訊號遮罩。

機制一 3. SIGCHLD 與殭屍行程（Zombie）的正確回收

當子行程呼叫 exit(0) 結束時，Linux 核心並不會立刻銷毀它的行程控制區塊（task_struct），而是保留其結束狀態碼（Exit Status）直到父行程呼叫 wait() 或 waitpid() 讀取為止；這段期間該子行程即成為殭屍行程（Zombie Process）。

為什麼 Handler 內部必須寫 while (waitpid(-1, NULL, WNOHANG) > 0)？

新手最常犯的錯誤，是在 on_sigchld 中只呼叫一次 waitpid(-1, NULL, WNOHANG)。這在並發環境下會漏收子行程，原因在於POSIX 標準訊號不支援排隊（Signal Coalescing）：

  • Pending 位元只有一個：核心對每種標準訊號（如 SIGCHLD）只用一個位元記錄「是否待處理」。若 3 個子行程在父行程執行 fork() 或 accept()（此時 SIGCHLD 處於 Blocked 狀態）期間同時結束，核心只會將 SIGCHLD 的 Pending 位元設為 1。
  • 解除封鎖後只觸發一次 Handler：當 pselect() 解除封鎖時，on_sigchld 只會被執行一次。若裡面沒有用 while 迴圈持續回收，就會有 2 個子行程永遠殘留為殭屍行程！
  • 保護 errno 不被破壞：waitpid() 在沒有更多已結束子行程時會回傳 -1 並將全域 errno 設為 ECHILD。若不先用 int saved_errno = errno; 備份並在離開前還原，就會覆蓋掉主程式正在檢查的系統呼叫錯誤碼。

完整實作 6. 改寫後完整程式碼與逐段解析（server4_clean.c）

以下為整合上述所有安全機制的完整 C 語言實作，可直接於 Linux / macOS / BSD 等任何 POSIX 相容系統編譯執行（gcc -std=c11 -Wall -Wextra -O2 server4_clean.c -o server4_clean）。

server4_clean.c — 具備 SIGCHLD 迴圈回收與 pselect() 原子保護的完整版本 
#define _POSIX_C_SOURCE 200809L

#include <sys/types.h>
#include <sys/socket.h>
#include <sys/wait.h>
#include <netinet/in.h>
#include <signal.h>
#include <errno.h>
#include <unistd.h>
#include <stdio.h>
#include <stdlib.h>

static volatile sig_atomic_t stop_server = 0;

/* 處理 SIGINT / SIGTERM：僅設定原子終止旗標，不在 Handler 內做複雜 I/O */
static void on_stop(int sig) {
    (void)sig;
    stop_server = 1;
}

/* 處理 SIGCHLD：以非阻塞迴圈回收所有已結束的子行程，並備份還原 errno */
static void on_sigchld(int sig) {
    (void)sig;
    int saved_errno = errno;
    while (waitpid(-1, NULL, WNOHANG) > 0) {
        /* 持續回收直到目前沒有已結束的子行程為止 */
    }
    errno = saved_errno;
}

int main(void) {
    /* 1. 使用 POSIX sigaction() 安裝訊號處理函式 */
    struct sigaction sa;
    sigemptyset(&sa.sa_mask);

    sa.sa_handler = on_sigchld;
    sa.sa_flags = SA_NOCLDSTOP; /* 子行程僅被暫停（SIGSTOP）時不觸發通知 */
    if (sigaction(SIGCHLD, &sa, NULL) == -1) {
        perror("sigaction(SIGCHLD)");
        return EXIT_FAILURE;
    }

    sa.sa_handler = on_stop;
    sa.sa_flags = 0;
    if (sigaction(SIGINT, &sa, NULL) == -1 || sigaction(SIGTERM, &sa, NULL) == -1) {
        perror("sigaction(SIGINT/SIGTERM)");
        return EXIT_FAILURE;
    }

    /* 2. 在進入主迴圈前先封鎖關鍵訊號，建立受保護的條件檢查區間 */
    sigset_t block_mask, orig_mask;
    sigemptyset(&block_mask);
    sigaddset(&block_mask, SIGCHLD);
    sigaddset(&block_mask, SIGINT);
    sigaddset(&block_mask, SIGTERM);
    if (sigprocmask(SIG_BLOCK, &block_mask, &orig_mask) == -1) {
        perror("sigprocmask");
        return EXIT_FAILURE;
    }

    /* 3. 建立、設定 SO_REUSEADDR 並綁定 TCP Listening Socket */
    int server_sockfd = socket(AF_INET, SOCK_STREAM, 0);
    if (server_sockfd == -1) {
        perror("socket");
        return EXIT_FAILURE;
    }

    int opt = 1;
    setsockopt(server_sockfd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    struct sockaddr_in server_address = {
        .sin_family = AF_INET,
        .sin_addr.s_addr = htonl(INADDR_ANY),
        .sin_port = htons(9734)
    };

    if (bind(server_sockfd, (struct sockaddr *)&server_address, sizeof(server_address)) == -1) {
        perror("bind");
        close(server_sockfd);
        return EXIT_FAILURE;
    }

    if (listen(server_sockfd, 128) == -1) {
        perror("listen");
        close(server_sockfd);
        return EXIT_FAILURE;
    }

    /* 4. 主事件迴圈：在訊號封鎖下檢查旗標，透過 pselect() 原子解除封鎖並等待 */
    while (!stop_server) {
        fd_set readfds;
        FD_ZERO(&readfds);
        FD_SET(server_sockfd, &readfds);

        printf("server waiting\n");
        int ready = pselect(server_sockfd + 1, &readfds, NULL, NULL, NULL, &orig_mask);
        if (ready == -1) {
            if (errno == EINTR) {
                continue; /* 被 SIGCHLD 或 SIGINT/SIGTERM 喚醒，回到頂端檢查 stop_server */
            }
            perror("pselect");
            break;
        }

        struct sockaddr_in client_address;
        socklen_t client_len = sizeof(client_address);
        int client_sockfd = accept(server_sockfd, (struct sockaddr *)&client_address, &client_len);
        if (client_sockfd == -1) {
            if (errno == EINTR) continue;
            perror("accept");
            continue;
        }

        pid_t pid = fork();
        if (pid == -1) {
            perror("fork");
            close(client_sockfd);
        } else if (pid == 0) {
            /* 子行程：關閉不需要的監聽 Socket，還原預設訊號與初始遮罩 */
            close(server_sockfd);
            signal(SIGINT, SIG_DFL);
            signal(SIGTERM, SIG_DFL);
            signal(SIGCHLD, SIG_DFL);
            sigprocmask(SIG_SETMASK, &orig_mask, NULL);

            char ch;
            if (read(client_sockfd, &ch, 1) == 1) {
                sleep(5);
                ch++;
                if (write(client_sockfd, &ch, 1) == -1) {
                    perror("write");
                }
            }
            close(client_sockfd);
            exit(EXIT_SUCCESS);
        } else {
            /* 父行程：關閉已交由子行程負責的連線 Socket */
            close(client_sockfd);
        }
    }

    printf("\nShutting down server...\n");
    close(server_sockfd);
    return EXIT_SUCCESS;
}

重點程式碼逐段解析

#define _POSIX_C_SOURCE 200809L
在標準 C11（-std=c11）模式下啟用 POSIX.1-2008 系統呼叫定義，確保編譯器正確宣告 sigaction、sigprocmask 與 pselect 等結構與函式原型。
static volatile sig_atomic_t
sig_atomic_t 保證單一指令即可完成讀寫（不會讀到寫一半的撕裂值），volatile 則禁止編譯器把 while (!stop_server) 最佳化快取成永遠為真的無窮迴圈。
SO_REUSEADDR 選項
允許伺服器在重啟時立即綁定仍處於 TIME_WAIT 狀態的 9734 埠號，避免開發測試與服務重啟時頻繁遭遇 bind: Address already in use 錯誤。
if (errno == EINTR) continue;
當子行程結束送出 SIGCHLD 或管理員按下 Ctrl+C 送出 SIGINT 時，pselect() 會被喚醒並回傳 -1（errno == EINTR）。透過 continue 回到迴圈頂端重新檢查 !stop_server，若是 SIGCHLD 就繼續下一輪等待，若是 SIGINT 則立刻跳出迴圈執行關機。
延伸思考：何時該用多行程（fork），何時該用單行程 I/O 多工（epoll）？

本篇示範的「pselect() + fork()」模型具備極佳的錯誤隔離性（單一子行程崩潰不會影響主伺服器與其他連線），非常適合連線數在數百以內、且每個請求需要執行獨立阻塞運算或外部程式的場景；若需要同時支撐數萬個高併發長連線，則可改用單行程非阻塞的 epoll 搭配 signalfd() 架構。
